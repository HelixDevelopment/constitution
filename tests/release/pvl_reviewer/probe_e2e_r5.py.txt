"""Reviewer probe r5: end-to-end CLI probes against a fixture root (bypass + robustness + DB write)."""
import hashlib, os, sqlite3, subprocess, sys
repo, work = sys.argv[1], sys.argv[2]
sys.path.insert(0, repo + "/tests/release")
import pvl_fixture as F
fx = F.make_root(work); root = fx["root"]
CK = repo + "/scripts/validation/check_provisional_verification_lines.py"
GOOD = F.line()
OPEN = ("CON-777", "In progress", "2026-11-01", "x")
def mkdb(path, rows, wal=False, created_type=str):
    if os.path.exists(path): os.remove(path)
    con = sqlite3.connect(path)
    if wal: con.execute("pragma journal_mode=wal")
    con.execute("create table items (atm_id text primary key, status text, body_md, created_at)")
    for r in rows: con.execute("insert into items values (?,?,?,?)", r)
    con.commit(); con.close()
def run(db, *extra):
    p = subprocess.run([sys.executable, "-I", CK, "--root", root, "--today", "2026-11-16", *extra, db],
                       capture_output=True, text=True, timeout=120)
    out = (p.stdout + p.stderr).strip().splitlines()
    verdict = [l for l in out if l.startswith("VERDICT") or l.startswith("FAIL ") or "Error" in l or "Traceback" in l]
    return p.returncode, verdict
def case(name, body, expect, created="2026-11-01", status="Ready for testing", **kw):
    db = os.path.join(work, "p.db")
    mkdb(db, [OPEN, ("CON-900", status, body, created)])
    rc, v = run(db)
    flag = "AS-EXPECTED" if rc == expect else "UNEXPECTED"
    print("%-12s rc=%s want=%s %s | %s" % (flag, rc, expect, name, " | ".join(v)[:260]))
# 1 lone CR hides the accepted line inside a CommonMark code block / comment (markdown-it renders it hidden)
case("CR: grammar line hidden in fenced code", "x\r```\n\n" + GOOD + "\n```\n", 1)
case("CR: grammar line hidden in comment", "x\r<!--\n\n" + GOOD + "\n-->\n", 1)
prose = "**Verification:** provisional -- job: not scheduled yet"
case("CR: visible prose line, hidden grammar line", "x\r" + prose + "\r```\n\n" + GOOD + "\n```\n", 1)
# 2 robustness of the database read

          sorted(k for k in after if before.get(k) != after[k]), " | ".join(v)[:160]))
    if holder: holder.close()
