#!/usr/bin/env python3
"""Survey: every field value of every verification line in the real DB (any status)."""
import importlib.util, sqlite3, sys
repo = sys.argv[1]
spec = importlib.util.spec_from_file_location("pvl", repo + "/scripts/validation/check_provisional_verification_lines.py")
m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m)
c = sqlite3.connect("file:%s/docs/workable_items.db?mode=ro" % repo, uri=True)
for i, s, b in c.execute("select atm_id,status,body_md from items order by atm_id"):
    for ln in m.unfenced_lines(b):
        if m.PREFIX_RE.match(ln):
            for part in m.SEP.split(ln)[1:]:
                mm = m.FIELD_PART.match(part)
                if mm:
                    v = mm.group(2).strip()
                    print("%s\t%s\t%s\t%s\t%s" % (i, s, mm.group(1), "A" if v.isascii() else "U", v[:160]))
