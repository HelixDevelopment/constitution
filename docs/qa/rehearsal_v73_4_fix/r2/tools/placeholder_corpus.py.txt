"""Corpus census of placeholder shapes in this checkout (tracked Markdown + tracker DB bodies).
Prints count per shape and a control needle line. Usage: python3 -I placeholder_corpus.py REPO"""
import re, sqlite3, subprocess, sys, collections
repo = sys.argv[1]
files = subprocess.run(["git", "-C", repo, "ls-files", "*.md"], capture_output=True, text=True, check=True).stdout.split()
texts = []
for f in files:
    try: texts.append(open("%s/%s" % (repo, f), encoding="utf-8", errors="replace").read())
    except OSError: pass
con = sqlite3.connect("file:%s/docs/workable_items.db?mode=ro" % repo, uri=True)
texts += [b or "" for (b,) in con.execute("select body_md from items")]
shapes = {
 "TBD": r"\bTBD\b", "TBA": r"\bTBA\b", "TODO": r"\bTODO\b", "FIXME": r"\bFIXME\b", "XXX": r"\bXXX\b",
 "UNCONFIRMED:": r"\bUNCONFIRMED:", "UNKNOWN": r"\bUNKNOWN\b", "PENDING": r"\bPENDING\b", "N/A": r"\bN/A\b",
 "angle <word>": r"<[a-z][a-z _-]{0,40}>", "mustache {{x}}": r"\{\{[^}]{1,40}\}\}", "shell ${X}": r"\$\{[A-Z_]+\}",
 "to be decided/determined/confirmed": r"(?i)\bto be (decided|determined|confirmed|defined|scheduled|announced)\b",
 "placeholder": r"(?i)\bplaceholder\b", "???": r"\?\?\?", "ellipsis value ': ...'": r":\s*(\.\.\.|…)\s*(--|—|$)",
 "dash value ': -'": r":\s*[-—–]\s*(--|—|$)",
}
c = collections.Counter()
for t in texts:
    for n, p in shapes.items():
        c[n] += len(re.findall(p, t, re.M))
print("corpus: %d markdown files + %d tracker bodies" % (len(files), len(texts) - len(files)))
for n in shapes: print("%-40s %d" % (n, c[n]))
needle = "zz-needle-TBD-7f3a"
print("CONTROL positive needle found:", bool(re.search(shapes["TBD"], needle)))
print("CONTROL negative (fabricated token QQZZ) count:", sum(len(re.findall(r"\bQQZZ\b", t)) for t in texts))
