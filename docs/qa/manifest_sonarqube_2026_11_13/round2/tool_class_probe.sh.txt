#!/usr/bin/env bash
# class_probe.sh <clone> <gate-relpath> <mode: dir|index>
# Runs one gate with an ambient git variable aimed at a fresh sentinel repo and
# reports whether the sentinel changed. Read-only towards everything except
# its own mktemp sentinel. No deletion.
set -u
clone="$1"; gate="$2"; mode="$3"
base="$(dirname "$0")"
SEN="$(mktemp -d "$base/sen.XXXXXX")"
gi() { env -u GIT_DIR -u GIT_INDEX_FILE -u GIT_WORK_TREE GIT_CONFIG_NOSYSTEM=1 GIT_CONFIG_GLOBAL=/dev/null git -C "$SEN" -c user.email=s@s -c user.name=s -c core.hooksPath=/dev/null "$@"; }
gi init -q -b sentinelbranch
printf 'sentinel\n' >"$SEN/keep.txt"
gi add keep.txt; gi commit -q -m sentinel
fp() { (cd "$SEN" && find . -path ./.git/logs -prune -o -type f -print0 | LC_ALL=C sort -z | xargs -0 sha256sum; find . -printf '%P %y %m %s\n' | LC_ALL=C sort) | sha256sum | cut -d' ' -f1; }
state() { gi for-each-ref --format='%(refname) %(objectname:short)'; gi log --all --oneline | head -5; gi config --local --list | LC_ALL=C sort; gi ls-files -s | head -5; }
b="$(fp)"; state >"$SEN.before"
case "$mode" in
  dir)   envs=(GIT_DIR="$SEN/.git") ;;
  index) envs=(GIT_INDEX_FILE="$SEN/.git/index") ;;
esac
s=$(date +%s)
rc=0; (cd "$clone" && env "${envs[@]}" timeout 1500 bash "$gate" >"$SEN.gate.out" 2>&1) || rc=$?
a="$(fp)"; state >"$SEN.after"
echo "gate=$gate mode=$mode env=${envs[*]} gate_rc=$rc wall=$(( $(date +%s)-s ))s"
if [ "$b" = "$a" ]; then echo "SENTINEL: unchanged (fingerprint $b)"; else echo "SENTINEL: CHANGED (before $b after $a)"; echo "--- state diff (before/after):"; diff "$SEN.before" "$SEN.after"; fi
echo "--- gate output tail:"; tail -n 6 "$SEN.gate.out"
