#!/usr/bin/env bash
# repro_pr_path.sh <source repo> <out dir> : CON-443 reproduction of the pull-request landing path.
# Builds a local bare "forge" from <source repo> main, opens a PR branch whose README badge row is
# stale (Anchors count edited), then lands it the way a §11.4.297 PR merge lands (merge commit,
# second parent = reviewed head, pushed to the forge main). Step "gate": if the repository ships a
# PR-side landing check (scripts/release/pr_landing_check.sh) it is run first and its refusal stops
# the merge. Result line: LANDED-STALE (RED: a stale badge row reached forge main), REFUSED (GREEN:
# the PR path refused), LANDED-CLEAN (a fresh row landed). Control needle: the badge check on the
# unmodified base must exit 0 and on the stale head must exit 1, else the instrument is blind.
set -u
for _ge_v in GIT_DIR GIT_WORK_TREE GIT_INDEX_FILE GIT_PREFIX GIT_COMMON_DIR GIT_OBJECT_DIRECTORY GIT_ALTERNATE_OBJECT_DIRECTORIES GIT_NAMESPACE GIT_SHALLOW_FILE GIT_GRAFT_FILE GIT_REPLACE_REF_BASE GIT_NO_REPLACE_OBJECTS GIT_QUARANTINE_PATH GIT_INTERNAL_SUPER_PREFIX; do unset "$_ge_v"; done
SRC=${1:?source repo}; OUT=${2:?out dir}; MODE=${3:-stale}
mkdir -p "$OUT"; W=$(mktemp -d "${TMPDIR:?}/reproPR.XXXXXX"); : "${W:?}"
export GIT_AUTHOR_NAME=repro GIT_AUTHOR_EMAIL=repro@invalid GIT_COMMITTER_NAME=repro GIT_COMMITTER_EMAIL=repro@invalid
export GIT_AUTHOR_DATE='2026-11-16T00:00:00Z' GIT_COMMITTER_DATE='2026-11-16T00:00:00Z' GIT_CONFIG_NOSYSTEM=1 GIT_CONFIG_GLOBAL=/dev/null
git clone -q --bare "$SRC" "$W/forge.git"
git clone -q "$W/forge.git" "$W/dev"; cd "$W/dev" || exit 2
base=$(git rev-parse HEAD); echo "base=$base"
bash scripts/badges/compute_badges.sh --root "$PWD" --check > "$OUT/control_base.txt" 2>&1; cb=$?
echo "control_base_rc=$cb"
git checkout -q -b pr/stale-badge
# a real PR change (a doc line) plus the drift: the README badge row is not regenerated
printf '\nRepro line for CON-443.\n' >> docs/qa/pr_path_drift_2026_11_16/README.md 2>/dev/null || { mkdir -p docs/qa/pr_path_drift_2026_11_16; printf 'Repro line for CON-443.\n' > docs/qa/pr_path_drift_2026_11_16/README.md; }
if [ "$MODE" = stale ]; then sed -i '3s#badge/anchors-\([0-9]*\)-green#badge/anchors-999-green#' README.md; fi
git add -A; git commit -q -m "docs: repro PR change" -m "Roadmap-Item: CON-443"
head=$(git rev-parse HEAD); echo "head=$head"
bash scripts/badges/compute_badges.sh --root "$PWD" --check > "$OUT/control_head.txt" 2>&1; ch=$?
echo "control_head_rc=$ch"
git push -q origin pr/stale-badge
# PR landing: optional PR-side check, then a merge commit on main (second parent = reviewed head)
git checkout -q main
gate=absent grc=-
if [ -f scripts/release/pr_landing_check.sh ] || git cat-file -e "$head:scripts/release/pr_landing_check.sh" 2>/dev/null; then
  gate=present
  bash "$(git rev-parse --show-toplevel)/scripts/release/pr_landing_check.sh" --repo "$W/dev" --base "$base" --head "$head" --evidence-dir "$OUT/prcheck" > "$OUT/prcheck.out" 2>&1; grc=$?
fi
echo "pr_landing_check=$gate rc=$grc"
if [ "$gate" = present ] && [ "$grc" -ne 0 ]; then echo "RESULT REFUSED"; rm -rf -- "${W:?}"; exit 0; fi
git merge -q --no-ff -m "Merge pull request #0 from pr/stale-badge" "$head"
git push -q origin main
git clone -q "$W/forge.git" "$W/after"
fmain=$(git -C "$W/after" rev-parse HEAD); echo "forge_main=$fmain parent2=$(git -C "$W/after" rev-parse HEAD^2)"
bash "$W/after/scripts/badges/compute_badges.sh" --root "$W/after" --check > "$OUT/forge_main_check.txt" 2>&1; fm=$?
echo "forge_main_badge_check_rc=$fm"
if [ "$fm" -ne 0 ]; then echo "RESULT LANDED-STALE"; else echo "RESULT LANDED-CLEAN"; fi
rm -rf -- "${W:?}"
