#!/usr/bin/env python3
"""mk_r2_behaviour_mutants.py <tree> <id>: round-2 mutants of the FIXED tree (N1..N5); each edit must
apply exactly once, else exit 2 (a no-op mutant)."""
import os, re, sys
root, mid = sys.argv[1], sys.argv[2]
P = lambda p: os.path.join(root, p)
def edit(p, old, new):
    t = open(P(p), encoding="utf-8").read()
    if t.count(old) != 1:
        sys.exit("NO-OP %s: %s x%d" % (mid, p, t.count(old)))
    open(P(p), "w", encoding="utf-8").write(t.replace(old, new)); print(mid, p)
SCRUB = re.compile(r'^for _ge_v in [^\n]*git-env-hermetic[^\n]*\n', re.M)
def drop_scrub(p):
    t = open(P(p), encoding="utf-8").read()
    if len(SCRUB.findall(t)) != 1:
        sys.exit("NO-OP %s: %s" % (mid, p))
    open(P(p), "w", encoding="utf-8").write(SCRUB.sub("", t)); print(mid, p)
if mid == "N1":   # the installer's top scrub clears the ceiling again (the F1 regression)
    edit("scripts/install_constitution.sh", "GIT_NAMESPACE GIT_SHALLOW_FILE", "GIT_NAMESPACE GIT_CEILING_DIRECTORIES GIT_SHALLOW_FILE")
elif mid == "N2":  # the canonical list holds the ceiling again
    edit("scripts/lib/constitution_layout.sh", 'GIT_NAMESPACE GIT_SHALLOW_FILE GIT_GRAFT_FILE GIT_REPLACE_REF_BASE GIT_NO_REPLACE_OBJECTS GIT_QUARANTINE_PATH GIT_INTERNAL_SUPER_PREFIX"',
         'GIT_NAMESPACE GIT_CEILING_DIRECTORIES GIT_SHALLOW_FILE GIT_GRAFT_FILE GIT_REPLACE_REF_BASE GIT_NO_REPLACE_OBJECTS GIT_QUARANTINE_PATH GIT_INTERNAL_SUPER_PREFIX"')
elif mid == "N3":  # F2 member loses its new scrub
    drop_scrub("scripts/fastcycle/tests/test_anchor_citations_red.sh")
elif mid == "N4":  # F2 wrapper member loses its new scrub (its own partial unset stays)
    drop_scrub("scripts/fastcycle/tests/test_custody_sweep_r15_regression.sh")
elif mid == "N5":  # the manifest gate clears the ceiling again (gate and its mutation test alike)
    for p in ("scripts/gates/cm_full_suite_manifest_complete.sh", "scripts/gates/cm_full_suite_manifest_complete_mutation_test.sh"):
        edit(p, "GIT_COMMON_DIR GIT_NAMESPACE\n", "GIT_COMMON_DIR GIT_NAMESPACE GIT_CEILING_DIRECTORIES\n")
else:
    sys.exit("unknown mutant")
