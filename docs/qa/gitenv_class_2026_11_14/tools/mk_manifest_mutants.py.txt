#!/usr/bin/env python3
"""mk_manifest_mutants.py <gate> <outdir>: writes one mutant copy of the manifest gate per entry, each a
single textual change, and refuses (exit 2) when a change does not apply exactly once (a no-op mutant)."""
import os, sys
gate, out = sys.argv[1:3]
src = open(gate, encoding="utf-8").read()
M = {
    # N1: the private XDG home removed from pgit and penv (the caller's default files are read again)
    "mN1_xdg": [('XDG_CONFIG_HOME="$XDG0" GIT_ATTR_NOSYSTEM=1 GIT_CONFIG_NOSYSTEM=1 GIT_CONFIG_GLOBAL=/dev/null git', 'GIT_ATTR_NOSYSTEM=1 GIT_CONFIG_NOSYSTEM=1 GIT_CONFIG_GLOBAL=/dev/null git'),
                ('penv() { env XDG_CONFIG_HOME="$XDG0" GIT_ATTR_NOSYSTEM=1', 'penv() { env GIT_ATTR_NOSYSTEM=1')],
    # N1: only the runner/export env (penv) loses the XDG home; pgit keeps its -c settings
    "mN1_penv_only": [('penv() { env XDG_CONFIG_HOME="$XDG0" GIT_ATTR_NOSYSTEM=1', 'penv() { env GIT_ATTR_NOSYSTEM=1')],
    # N1: GIT_ATTR_NOSYSTEM dropped everywhere (needs a system attributes file to observe)
    "mN1_nosystem": [('XDG_CONFIG_HOME="$XDG0" GIT_ATTR_NOSYSTEM=1 GIT_CONFIG_NOSYSTEM=1 GIT_CONFIG_GLOBAL=/dev/null git', 'XDG_CONFIG_HOME="$XDG0" GIT_CONFIG_NOSYSTEM=1 GIT_CONFIG_GLOBAL=/dev/null git'),
                     ('penv() { env XDG_CONFIG_HOME="$XDG0" GIT_ATTR_NOSYSTEM=1', 'penv() { env XDG_CONFIG_HOME="$XDG0"')],
    # rv01: `manifest missing:` no longer a FAIL class
    "rv01": [('"manifest missing: "*|"manifest: "*', '"manifest: "*')],
    # rv03: alternates points at $ROOT/.git/objects instead of the real object directory
    "rv03": [("printf '%s\\n' \"$OBJ\" >\"$S/.git/objects/info/alternates\"", "printf '%s\\n' \"$ROOT/.git/objects\" >\"$S/.git/objects/info/alternates\"")],
    # rv04: rgit without -c core.fsmonitor=false
    "rv04": [('rgit() { git --literal-pathspecs --no-optional-locks -c core.fsmonitor=false -C', 'rgit() { git --literal-pathspecs --no-optional-locks -C')],
}
os.makedirs(out, exist_ok=True)
for name, edits in M.items():
    t = src
    for old, new in edits:
        if t.count(old) != 1:
            print("NO-OP or ambiguous mutant %s: %r occurs %d times" % (name, old[:50], t.count(old))); sys.exit(2)
        t = t.replace(old, new)
    p = os.path.join(out, name + ".sh")
    open(p, "w", encoding="utf-8").write(t)
    os.chmod(p, 0o755)
    print(p)
