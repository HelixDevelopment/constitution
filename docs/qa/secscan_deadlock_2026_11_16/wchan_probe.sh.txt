#!/usr/bin/env bash
# wchan_probe.sh SCRIPT -- run SCRIPT's exact-match pass on a big hermetic repo,
# wait 25 s, then print the kernel wait channel of every descendant (own tree only).
set -uo pipefail
for _ge_v in GIT_DIR GIT_WORK_TREE GIT_INDEX_FILE GIT_PREFIX GIT_COMMON_DIR GIT_OBJECT_DIRECTORY GIT_ALTERNATE_OBJECT_DIRECTORIES GIT_NAMESPACE GIT_SHALLOW_FILE GIT_GRAFT_FILE GIT_REPLACE_REF_BASE GIT_NO_REPLACE_OBJECTS GIT_QUARANTINE_PATH GIT_INTERNAL_SUPER_PREFIX; do unset "$_ge_v"; done; unset _ge_v  # git-env-hermetic: clear the caller's repository-locating git variables first (list = CL_GIT_SCRUB_VARS of scripts/lib/constitution_layout.sh; checked by scripts/validation/git_env_class_census.py)
S="${1:?script}"; SRC="${2:?src root}"
export TMPDIR=/mnt/DATA1TB/claude-work/tmp
t=$(mktemp -d); trap 'rm -rf "${t:?}"' EXIT
export GIT_AUTHOR_NAME=t GIT_AUTHOR_EMAIL=t@t GIT_COMMITTER_NAME=t GIT_COMMITTER_EMAIL=t@t SEC_TEST_LOG="$t"
git init -q "$t/big"
python3 -I -c '
import sys
o=sys.stdout.buffer
o.write(b"commit refs/heads/main\ncommitter t <t@t> 946684800 +0000\ndata 3\nbig\n")
for i in range(8000):
    b=("file %06d filler %d\n"%(i,i*7919)).encode()
    o.write(b"M 100644 inline f/%06d.txt\ndata %d\n"%(i,len(b))+b+b"\n")' | git -C "$t/big" fast-import --quiet
python3 -I -c 'import secrets;print("export KEY_A=kv"+secrets.token_hex(12))' > "$t/k.sh"; chmod 600 "$t/k.sh"
mkdir "$t/scr"
env PATH="$SRC/tests/security/fixtures/bin:$PATH" SAFE_RM="$SRC/scripts/lib/safe_rm.sh" SCRATCH="$t/scr" OUT="$t/scr/o.tsv" bash "$S" --keys-file "$t/k.sh" "$t/big" >"$t/out" 2>"$t/err" &
top=$!
sleep 25
desc(){ local p=$1 c; echo "$p"; for c in $(cat /proc/"$p"/task/*/children 2>/dev/null); do desc "$c"; done; }
for p in $(desc "$top"); do
  [ -r /proc/"$p"/cmdline ] || continue
  cmd=$(tr '\0' ' ' < /proc/"$p"/cmdline | cut -c1-60)
  printf '%s\t%s\t%s\t%s\n' "$p" "$(cat /proc/"$p"/wchan 2>/dev/null)" "$(awk '/^State/{print $2}' /proc/"$p"/status)" "$cmd"
done
alive=no; kill -0 "$top" 2>/dev/null && alive=yes
echo "script still running after 25s: $alive"
# stop only our own tree (children first), never a group
for p in $(desc "$top" | tac); do kill "$p" 2>/dev/null; done
wait "$top" 2>/dev/null; echo "stdout of run:"; cat "$t/out"; echo "stderr tail:"; tail -3 "$t/err" | cut -c1-120
