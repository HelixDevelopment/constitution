import os, re, sys
root = sys.argv[1]; entries = sys.argv[2:]
REF = re.compile(r"[A-Za-z0-9_./${}-]*?([A-Za-z0-9_.-]+\.(?:sh|py|bash))\b")
def files_by_base():
    d = {}
    for dp, dn, fn in os.walk(root):
        dn[:] = [x for x in dn if x not in (".git", "submodules", "node_modules")]
        for f in fn:
            if f.endswith((".sh", ".py", ".bash")):
                d.setdefault(f, []).append(os.path.relpath(os.path.join(dp, f), root))
    return d
B = files_by_base()
seen = {}; todo = list(entries)
while todo:
    f = todo.pop()
    if f in seen: continue
    seen[f] = []
    try: lines = open(os.path.join(root, f), encoding="utf-8", errors="replace").read().splitlines()
    except OSError: continue
    for line in lines:
        if line.lstrip().startswith("#"): continue
        for m in REF.finditer(line):
            tok = m.group(0); base = m.group(1)
            cands = B.get(base, [])
            if not cands: continue
            pick = None
            if "scripts/" in tok:
                suf = tok[tok.index("scripts/"):]
                pick = [c for c in cands if c.endswith(suf.split("}")[-1]) or c == suf]
            if not pick:
                same = [c for c in cands if os.path.dirname(c) == os.path.dirname(f)]
                pick = same or cands
            for c in pick:
                if c.startswith("tests/") or "/tests/" in c: continue
                seen[f].append(c)
                if c not in seen: todo.append(c)
for f in sorted(seen): print(f)
