Each mutant: a detached git worktree of COMMIT under WORKROOT, exact replacements (each OLD must occur exactly the
stated number of times), then the named test runs there; KILLED when the test exits non-zero. A control run on the
unmutated worktree must exit 0 first. Worktrees are removed with `git worktree remove --force`. At most two run at
a time (each under systemd-run --user --scope -p TasksMax=2000 nice -n 10).

def sh(args, cwd=None, log=None):
    return subprocess.run(args, cwd=cwd, stdout=log or subprocess.PIPE, stderr=subprocess.STDOUT, text=True)


def one(repo, commit, root, case):
    mid, desc, edits, test = case
    wt = os.path.join(root, mid)
    r = sh(["git", "-C", repo, "worktree", "add", "-q", "--detach", wt, commit])
    if r.returncode: return [(mid, desc, "-", "WORKTREE-FAILED", r.stdout.strip()[:160])]
    for f, old, new, n in edits:
        p = os.path.join(wt, f); s = open(p, encoding="utf-8").read()
        if s.count(old) != n:
            sh(["git", "-C", repo, "worktree", "remove", "--force", wt])
            return [(mid, desc, "-", "ANCHOR-MISSING %s count %d" % (f, s.count(old)), "")]
        open(p, "w", encoding="utf-8").write(s.replace(old, new))
    lp = os.path.join(root, "%s__%s.log" % (mid, os.path.basename(test)))
    t0 = time.time()
    with open(lp, "w") as lf:
        rc = sh(["systemd-run", "--user", "--scope", "-q", "-p", "TasksMax=2000", "nice", "-n", "10", "bash", test], cwd=wt, log=lf).returncode
    first = ""
    for line in open(lp, encoding="utf-8", errors="replace"):
        if line.startswith(("  FAIL", "  BLIND", "FAIL", "BLIND")): first = line.strip()[:200]; break
    sh(["git", "-C", repo, "worktree", "remove", "--force", wt])
    if mid == "CONTROL": verdict = "PASS" if rc == 0 else "CONTROL-FAILED"
    else: verdict = "KILLED" if rc != 0 else "SURVIVED"
    return [(mid, desc, os.path.basename(test), "%s rc=%d %ds" % (verdict, rc, round(time.time() - t0)), first)]


def main():
    repo, commit, root, out = sys.argv[1:5]
    want = set(sys.argv[5:]); root = os.path.abspath(root); os.makedirs(root, exist_ok=True)
    rows = one(repo, commit, root, ("CONTROL", "unmutated", [], TT))
    print("\t".join(rows[0])); sys.stdout.flush()
    if not rows[0][3].startswith("PASS"): open(out, "w").write("\t".join(rows[0]) + "\n"); sys.exit(1)
    cases = [m for m in MUT if not want or m[0] in want]
    with concurrent.futures.ThreadPoolExecutor(max_workers=2) as ex:
        for res in ex.map(lambda c: one(repo, commit, root, c), cases):
            for r in res: rows.append(r); print("\t".join(map(str, r))); sys.stdout.flush()
    with open(out, "w") as f:
        for r in rows: f.write("\t".join(map(str, r)) + "\n")


main()
