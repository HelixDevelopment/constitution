# round-14 fixer mutants: tools/mut14.txt blocks "@@@ <name> <file>\n<old>===\n<new>>>>"
_t = open(os.path.join(os.path.dirname(os.path.abspath(__file__)), "mut14.txt")).read()
for _blk in _t.split("@@@ ")[1:]:
    _hdr, _rest = _blk.split("\n", 1)
    _name, _file = _hdr.split()
