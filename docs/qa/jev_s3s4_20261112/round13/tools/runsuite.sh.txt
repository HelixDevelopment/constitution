#!/bin/bash
# runsuite.sh <repo> <suite-key> <out-prefix> [producer]
set -u
R=/mnt/DATA1TB/claude-work/scratch/jev_r13.7MaT
REPO="${1:?}"; K="${2:?}"; O="${3:?}"; P="${4:-$REPO/scripts/jev/jev-place.sh}"
case "$K" in r12) f=tests/jev/test_s4_round12.sh ;; r12mc) f=tests/jev/test_s4_round12_mc.sh ;; r13) f=tests/jev/test_s4_round13.sh ;; r11) f=tests/jev/test_s4_round11.sh ;; r10) f=tests/jev/test_s4_round10.sh ;; r9) f=tests/jev/test_s4_round9.sh ;; r8) f=tests/jev/test_s4_round8.sh ;; r7) f=tests/jev/test_s4_round7.sh ;; r6) f=tests/jev/test_s4_round6.sh ;; pl) f=tests/jev/test_s4_placement.sh ;; tests/*) f="$K" ;; *) exit 2 ;; esac
while [ "$(cut -d. -f1 /proc/loadavg)" -ge 60 ]; do sleep 20; done
st=$(date +%s)
HRUN_EXTRA_ENV="S4_PLACE_UNDER_TEST=$P" systemd-run --user --scope -q -p TasksMax=2000 nice -n 10 \
  "$R/tools/hrun.sh" "$REPO" timeout --kill-after=10 2400 bash "$f" > "$O.txt" 2>&1
rc=$?
nf=$(grep -c '^FAIL\|^not ok' "$O.txt"); np=$(grep -c '^PASS\|^ok' "$O.txt")
printf '%s\t%s\trc=%s\tfails=%s\tpasses=%s\t%ss\n' "$(basename "$O")" "$K" "$rc" "$nf" "$np" "$(( $(date +%s) - st ))" >> "$R/runs/results.tsv"
