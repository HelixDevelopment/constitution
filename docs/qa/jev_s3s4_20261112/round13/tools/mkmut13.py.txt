    d = os.path.join(md, name)
    if os.path.exists(d): shutil.rmtree(d)
    if name.startswith("C"):
        shutil.copytree(os.path.join(repo, "scripts/validation"), os.path.join(d, "scripts/validation"))
    else:
        shutil.copytree(os.path.join(repo, "scripts/jev"), os.path.join(d, "scripts/jev"))

        t = t.replace(sub[1], sub[2]); open(f, "w", encoding="utf-8").write(t)
