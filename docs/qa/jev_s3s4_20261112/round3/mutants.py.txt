def sh(*a, **k):
    return subprocess.run(list(a), cwd=REPO, capture_output=True, text=True, **k)

def run_suite(suite, log):
    env = dict(os.environ)
    tmp = os.path.join(OUT, "tmp"); os.makedirs(tmp, exist_ok=True); env["TMPDIR"] = tmp
    p = subprocess.Popen(["nice", "-n", "10", "timeout", "900", "bash", suite], cwd=REPO, env=env,
                         stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, start_new_session=True)
    first_fail, lines = None, []
    for line in p.stdout:
        lines.append(line)
        if first_fail is None and line.startswith("FAIL - "):
            first_fail = line.strip()
            pg = os.getpgid(p.pid)
            if isinstance(pg, int) and pg > 1 and pg == p.pid:   # our own new session only (§11.4.263)
                os.killpg(pg, signal.SIGTERM)
            break
    try:
        rc = p.wait(timeout=120)
    except subprocess.TimeoutExpired:
        pg = p.pid
        if pg > 1: os.killpg(pg, signal.SIGKILL)
        rc = p.wait()
    with open(log, "w") as f: f.writelines(lines)
    summary = next((l.strip() for l in reversed(lines) if l.startswith("# test_")), "")
    return rc, first_fail, summary

results = []
for mid, desc, suite, edits in M:
    if ONLY and mid not in ONLY: continue
    files = sorted({e[0] for e in edits})
    status = "INVALID"
    detail = ""
    try:
        for path, old, new in edits:
            fp = os.path.join(REPO, path)
            s = open(fp).read()
            n = s.count(old)
            if n != 1:
                detail = f"anchor count {n} in {path}"; raise ValueError(detail)
            open(fp, "w").write(s.replace(old, new))
        if os.environ.get("MUT_DRY"):
            status = "VALID"; raise ValueError("dry")
        t0 = time.time()
        rc, ff, summ = run_suite(suite, os.path.join(OUT, f"{mid}.log"))
        dt = int(time.time() - t0)
        if ff or rc != 0:
            status = "KILLED"; detail = (ff or f"rc={rc} {summ}")[:200]
        else:
            status = "SURVIVED"; detail = summ
        detail += f" ({dt}s)"
    except ValueError:
        pass
    finally:
        sh("git", "checkout", "--", *files)
        dirty = sh("git", "status", "--porcelain").stdout.strip()
        if dirty:
            print(f"ABORT: tree not clean after {mid}: {dirty}", flush=True); sys.exit(3)
    results.append({"id": mid, "desc": desc, "suite": os.path.basename(suite), "status": status, "detail": detail})
    print(f"{mid}\t{status}\t{os.path.basename(suite)}\t{desc}\t{detail}", flush=True)

with open(os.path.join(OUT, "results.json"), "a") as f:
    for r in results: f.write(json.dumps(r, ensure_ascii=False) + "\n")
