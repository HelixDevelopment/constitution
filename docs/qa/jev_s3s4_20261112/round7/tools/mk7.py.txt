import re
for i, f in enumerate(rows, 1):
    line = [l for l in orig.splitlines() if re.match(r"\s*(\[\s*)?\[\s*\"%s\",\s+ok\(" % re.escape(f), l)]
    assert len(line) == 1, (f, line)
    a = line[0]
    b = re.sub(r'ok\(.*\)\](,?)\s*(\]\')?$', lambda mm: 'true]' + mm.group(1) + (' ' + mm.group(2) if mm.group(2) else ''), a)
    assert b != a, f
    m("CT%02d" % i, a, b, "contract table row %s not checked" % f, [P72, R7, R6, PL])

bad = 0
with open(os.path.join(W, "mut", "index.tsv"), "w") as idx:
    for k, (a, b, desc, targets) in M.items():
        n = orig.count(a)
        if n != 1:
            print(f"{k}: pattern matched {n} times - refused", file=sys.stderr); bad += 1; continue
        mut = orig.replace(a, b, 1)
        assert mut != orig
        d = os.path.join(W, "mut", k)
        if os.path.isdir(d): shutil.rmtree(d)
        os.makedirs(os.path.join(d, "scripts"))
        shutil.copytree(SRC, os.path.join(d, "scripts", "jev"), symlinks=True)
        os.symlink(os.path.join(W, "repo", "submodules"), os.path.join(d, "submodules"))
        p = os.path.join(d, "scripts", "jev", "jev-place.sh")
        open(p, "w").write(mut); os.chmod(p, 0o755)
        idx.write(f"{k}\t{','.join(targets)}\t{hashlib.sha256(mut.encode()).hexdigest()}\t{desc}\n")
print(f"mutants: {len(M) - bad} ready, {bad} refused")
sys.exit(1 if bad else 0)
