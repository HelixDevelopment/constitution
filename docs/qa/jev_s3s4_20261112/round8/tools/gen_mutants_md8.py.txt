#!/usr/bin/env python3
"""gen_mutants_md8.py <round8 dir>: writes mutants.md from mutants/index.tsv and mutants/results.tsv (no hand edits)."""
import sys, os, collections
D = sys.argv[1]
idx = [l.rstrip("\n").split("\t") for l in open(os.path.join(D, "mutants", "index.tsv")) if l.strip()]
res = collections.defaultdict(list)
for l in open(os.path.join(D, "mutants", "results.tsv")):
    f = l.rstrip("\n").split("\t")
    if len(f) >= 7: res[f[0]].append(f)
EQ = {
 "NM04": "equivalent (round 7 argument, mutants.md of round 7): `leases_save` writes `gen` into the file before hashing, so a hash-only token already changes with every write",
 "OM13": "equivalent (round 7 argument; control OM13x killed in round 7): the explicit reason code at the refusal site makes the derived one unreachable",
 "RM09": "equivalent (round 7 argument): every writer of a `stopping` mark writes it on a set with no lease, so derive_cache never sees a leased stopping set",
 "OWN08": "equivalent (round-7 reviewer's argument): a foreign `starting` mark present at plan time is refused at plan (C); one written later changes the token, so seam 2 returns first",
}
out = ["# Jev S3/S4 round 8 — mutants", "", "| Field | Value |", "|---|---|", "| Revision | 1 |", "| Created | 2026-11-14 |",
       "| Last modified | 2026-11-14 |", "| Status | evidence (generated by `tools/gen_mutants_md8.py.txt`; no GO claimed) |",
       "| Evidence class | gate-logic only (stub llmctl, §11.4.289(C)(2)) |", "",
       "Every mutant is the round-8 producer with the edits of `tools/mk8.py.txt` (each must match exactly once; 0 refused), run hermetic",
       "(`tools/hrun.sh.txt`) through `S4_PLACE_UNDER_TEST`, suite by suite in the order of its targets, stopping at the first suite that",
       "kills it (`tools/runmut8.sh.txt`, two lanes, `tools/drive8.sh.txt`). REAL is the unmodified producer on all four suites.", ""]
cnt = collections.Counter(); rows = []
for k, targets, sha, desc in idx:
    r = res.get(k, [])
    runs = "; ".join(f"{x[1]}: {x[2]} ({x[6]})" for x in r) or "NO RUN"
    final = r[-1][2] if r else "no run"
    if final == "killed": v, why = "killed", r[-1][7] if len(r[-1]) > 7 else ""
    elif k in EQ: v, why = "equivalent", EQ[k]
    else: v, why = "SURVIVED (test gap)", ""
    cnt[v] += 1
    rows.append(f"| {k} | {desc.replace('|', '/')} | {runs} | {v} | {why.replace('|', '/')[:220]} |")
out += [f"Totals: {len(idx)} mutants: " + ", ".join(f"{n} {v}" for v, n in sorted(cnt.items())) + ".", "",
        "REAL (unmodified producer, no FAIL line, exit 0): " + "; ".join(f"{x[1]}: {'pass' if x[2] == 'survived' else 'FAIL'} ({x[5]}, {x[6]})" for x in res.get("REAL", [])) + ".", "",
        "| Key | Edit | Runs | Verdict | First failing check / argument |", "|---|---|---|---|---|"] + rows + [""]
out += ["The four equivalence arguments rest on code the round-8 change does not touch: its producer diff against `c811d0cd` has five",
        "hunks (the header Env line and MARK TTL paragraph, THE CLOCK block, the reap comment, and `clock_refresh` in `with_lock`).", ""]
open(os.path.join(D, "mutants.md"), "w").write("\n".join(out))
print(dict(cnt))
